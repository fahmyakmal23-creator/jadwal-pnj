main.cpp // C++11 Standard 
#include <iostream>
#include <string>
using namespace std;

// Struct untuk merepresentasikan satu jadwal mata kuliah PNJ
struct MataKuliah {
    int hariIdx;        // 0=Senin(Libur), 1=Selasa, 2=Rabu, 3=Kamis, 4=Jumat
    string matkul;
    string jam;
    string ruang;
    string status;      // "MASUK", "ONLINE", atau "KOSONG"
    string catatan;
};

int main() {
    const string PIN_PJ = "••••••••"; // Rahasia PJ Kelas

    string namaHari[7] = {
        "Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu", "Minggu"
    };

    // Daftar pilihan baku untuk Mode PJ Kelas
    string opsiMatkul[8] = {
        "Matematika Diskrit", "Jaringan Komputer", "Sistem Operasi",
        "Piranti Komputer", "Agama Islam", "Algoritma Dan Pemrograman",
        "Bahasa Indonesia", "Pengantar TIK"
    };
    string opsiJam[3] = {
        "08.00 - 13.20", "13.20 - 17.00", "11.00 - Selesai"
    };
    string opsiRuang[6] = {
        "AA301", "Perpus 8", "PUT TIK 2",
        "Perpus 2", "Perpus 1", "Online (Zoom / GMeet)"
    };

    // Array jadwal baku PNJ: Senin Libur (0 matkul), Selasa-Jumat (2 matkul/hari)
    MataKuliah jadwal[8] = {
        {1, "Matematika Diskrit", "08.00 - 13.20", "AA301", "MASUK", "Sesuai jadwal"},
        {1, "Jaringan Komputer", "13.20 - 17.00", "PUT TIK 2", "MASUK", "Praktikum lab"},
        {2, "Sistem Operasi", "08.00 - 13.20", "Perpus 8", "ONLINE", "Link Zoom dibagikan di grup WA"},
        {2, "Piranti Komputer", "13.20 - 17.00", "PUT TIK 2", "MASUK", "Sesuai jadwal"},
        {3, "Agama Islam", "08.00 - 13.20", "Perpus 2", "MASUK", "Sesuai jadwal"},
        {3, "Algoritma Dan Pemrograman", "13.20 - 17.00", "PUT TIK 2", "MASUK", "Presentasi studi kasus"},
        {4, "Bahasa Indonesia", "08.00 - 13.20", "Perpus 1", "MASUK", "Sesuai jadwal"},
        {4, "Pengantar TIK", "13.20 - 17.00", "AA301", "MASUK", "Sesuai jadwal"}
    };
    int jumlahData = 8;

    int todayIdx;
    cout << "=== jadwal.pnj // h-1 checker ===\n";
    cout << "Masukkan indeks hari ini (0=Sen .. 6=Min): ";
    cin >> todayIdx;

    if (todayIdx < 0 || todayIdx > 6) {
        cout << "Error: Indeks hari harus antara 0 sampai 6.\n";
        return 1;
    }

    // Rumus modulo untuk menghitung indeks hari besok (H+1)
    int besokIdx = (todayIdx + 1) % 7;
    cout << "Hari ini : " << namaHari[todayIdx] << " (" << todayIdx << ")\n";
    cout << "Besok    : " << namaHari[besokIdx] << " (" << besokIdx << ")\n";

    int menu;
    do {
        cout << "\nMenu Utama:\n";
        cout << "1. Cek Jadwal Besok (Mode Mahasiswa)\n";
        cout << "2. Update Jadwal & Status (Mode PJ Kelas)\n";
        cout << "0. Keluar\n";
        cout << "Pilih: ";
        cin >> menu;

        switch (menu) {
            case 1: {
                cout << "\n--- JADWAL BESOK: " << namaHari[besokIdx] << " ---\n";
                if (besokIdx == 0) {
                    cout << "LIBUR KULIAH // SENIN OFF (0 Matkul) — Tidak ada jadwal reguler pada hari Senin.\n";
                } else if (besokIdx == 5 || besokIdx == 6) {
                    cout << "LIBUR AKHIR PEKAN — Tidak ada perkuliahan reguler.\n";
                } else {
                    int ketemu = 0;
                    for (int i = 0; i < jumlahData; i++) {
                        if (jadwal[i].hariIdx == besokIdx) {
                            ketemu++;
                            cout << "\n[" << ketemu << "] " << jadwal[i].matkul
                                 << " [" << jadwal[i].status << "]\n";
                            cout << "    Jam     : " << jadwal[i].jam << "\n";
                            cout << "    Ruangan : " << jadwal[i].ruang << "\n";
                            cout << "    Catatan : " << jadwal[i].catatan << "\n";
                        }
                    }
                    if (ketemu == 0) {
                        cout << "Tidak ada jadwal tercatat untuk hari besok.\n";
                    }
                }
                break;
            }

            case 2: {
                string pin;
                cout << "Masukkan PIN PJ Kelas: ";
                cin >> pin;

                if (pin != PIN_PJ) {
                    cout << "Akses ditolak: PIN salah.\n";
                    break;
                }

                cout << "\nDaftar Jadwal Aktif (Selasa - Jumat):\n";
                for (int i = 0; i < jumlahData; i++) {
                    cout << i << ". [" << namaHari[jadwal[i].hariIdx] << "] "
                         << jadwal[i].matkul << " | " << jadwal[i].jam
                         << " | " << jadwal[i].ruang << " (" << jadwal[i].status << ")\n";
                }

                int idx;
                cout << "Pilih indeks jadwal (0.." << jumlahData - 1 << "): ";
                cin >> idx;

                if (idx >= 0 && idx < jumlahData) {
                    int pilJam, pilRuang, pilStatus;

                    cout << "\nPilih Jam Perkuliahan:\n";
                    for (int j = 0; j < 3; j++) {
                        cout << (j + 1) << ". " << opsiJam[j] << "\n";
                    }
                    cout << "Pilihan (1-3): ";
                    cin >> pilJam;
                    if (pilJam >= 1 && pilJam <= 3) jadwal[idx].jam = opsiJam[pilJam - 1];

                    cout << "\nPilih Status Kelas (1=MASUK, 2=ONLINE, 3=KOSONG): ";
                    cin >> pilStatus;
                    if (pilStatus == 2) {
                        jadwal[idx].status = "ONLINE";
                        jadwal[idx].ruang = "Online (Zoom / GMeet)";
                    } else if (pilStatus == 3) {
                        jadwal[idx].status = "KOSONG";
                    } else {
                        jadwal[idx].status = "MASUK";
                    }

                    cout << "\nPilih Ruangan Kampus:\n";
                    for (int r = 0; r < 6; r++) {
                        cout << (r + 1) << ". " << opsiRuang[r] << "\n";
                    }
                    cout << "Pilihan (1-6): ";
                    cin >> pilRuang;
                    if (pilRuang >= 1 && pilRuang <= 6) jadwal[idx].ruang = opsiRuang[pilRuang - 1];

                    cin.ignore();
                    cout << "Catatan PJ: ";
                    getline(cin, jadwal[idx].catatan);

                    cout << "Data jadwal berhasil diperbarui.\n";
                } else {
                    cout << "Indeks tidak valid.\n";
                }
                break;
            }

            case 0:
                cout << "Program selesai.\n";
                break;

            default:
                cout << "Pilihan menu tidak tersedia.\n";
        }
    } while (menu != 0);

    return 0;
}
